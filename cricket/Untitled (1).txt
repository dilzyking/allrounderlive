function showMessage(text){
  message.textContent = text;
  message.hidden = false;
}

function hideMessage(){
  message.hidden = true;
}

function safe(text){
  return String(text ?? '').replace(
    /[&<>"']/g,
    c => ({
      '&':'&amp;',
      '<':'&lt;',
      '>':'&gt;',
      '"':'&quot;',
      "'":'&#39;'
    }[c])
  );
}

function normalizeKey(key){
  if(!key) return {};

  if(typeof key === 'object' && !Array.isArray(key)){
    return Object.fromEntries(
      Object.entries(key)
        .filter(([k,v]) =>
          /^[a-f\d]{32}$/i.test(k) &&
          /^[a-f\d]{32}$/i.test(v)
        )
        .map(([k,v]) => [k.toLowerCase(), v.toLowerCase()])
    );
  }

  const result = {};

  String(key).split(/[;,\n]/).forEach(pair => {
    const match = pair.trim().match(
      /^([a-f\d]{32})\s*:\s*([a-f\d]{32})$/i
    );

    if(match){
      result[match[1].toLowerCase()] = match[2].toLowerCase();
    }
  });

  return result;
}

function getStreams(match){
  const collections = [
    match.dashStreams,
    match.hlsStreams,
    match.streams,
    match.event?.dashStreams,
    match.event?.hlsStreams
  ];

  const found = [];

  for(const collection of collections){
    if(!Array.isArray(collection)) continue;

    for(const item of collection){
      if(!item || typeof item !== 'object') continue;

      const url = String(
        item.url || item.streamUrl || item.src || ''
      ).trim();

      if(!/^https?:\/\//i.test(url)) continue;

      let type = String(item.type || '').toLowerCase();

      if(!type){
        type = /\.m3u8(?:\?|$)/i.test(url) ? 'hls' : 'dash';
      }

      if(type === 'mpd') type = 'dash';
      if(type === 'm3u8') type = 'hls';

      if(!['dash','hls'].includes(type)) continue;
      if(found.some(s => s.url === url)) continue;

      found.push({
        title: item.title || item.name || `Server ${found.length + 1}`,
        url,
        type,
        key: item.key || item.clearKey || item.clearkey || item.keys || null
      });
    }
  }

  return found;
}

function renderChannels(){
  if(!streams.length){
    grid.innerHTML = '<div class="channels-empty">No playable servers are available for this match.</div>';
    return;
  }

  grid.innerHTML = streams.map((s,i) => `
    <button
      type="button"
      class="channel-card ${i === activeIndex ? 'active' : ''}"
      data-index="${i}"
      aria-pressed="${i === activeIndex}"
    >
      <span class="icon">
        ${safe((s.title.match(/[A-Za-z0-9]/g) || []).slice(0,2).join('').toUpperCase() || 'HD')}
      </span>
      <span class="name">${safe(s.title)}</span>
      <span class="badge">
        ${s.type === 'hls' ? 'HLS' : 'DASH'} · ${i === activeIndex ? 'Playing' : 'HD'}
      </span>
    </button>
  `).join('');
}

function configureStreaming(){
  player.configure({
    streaming: {
      lowLatencyMode: true,
      bufferingGoal: 15,
      rebufferingGoal: 2,
      bufferBehind: 15,
      retryParameters: {
        timeout: 10000,
        maxAttempts: 5,
        baseDelay: 300,
        backoffFactor: 1.2
      },
      segmentRequestTimeout: 8000,
      segmentPrefetchLimit: 2,
      useNativeHlsOnSafari: true
    },
    manifest: {
      retryParameters: {
        timeout: 8000,
        maxAttempts: 3
      }
    }
  });
}

function switchChannel(index){
  const stream = streams[index];

  if(!stream || !player) return Promise.resolve();

  const requestId = ++switchId;

  activeIndex = index;
  renderChannels();

  isSwitching = true;
  hideMessage();
  showBuffer(0);

  switchQueue = switchQueue.catch(() => {}).then(async () => {
    if(requestId !== switchId) return;

    let loadTimer;

    try {
      video.pause();

      await player.unload();

      if(requestId !== switchId) return;

      player.resetConfiguration();

      configureStreaming();

      if(stream.type === 'dash'){
        player.configure({
          drm: {
            clearKeys: normalizeKey(stream.key)
          }
        });
      } else {
        player.configure({
          drm: {
            clearKeys: {}
          }
        });
      }

      const mimeType = stream.type === 'hls'
        ? 'application/x-mpegurl'
        : 'application/dash+xml';

      const timedLoad = Promise.race([
        player.load(stream.url, null, mimeType),

        new Promise((_,reject) => {
          loadTimer = setTimeout(
            () => reject(new Error('Stream load timed out')),
            LOAD_TIMEOUT
          );
        })
      ]);

      await timedLoad;

      if(requestId !== switchId) return;

      isSwitching = false;

      lastPosition = video.currentTime;
      lastProgressAt = performance.now();

      hideMessage();

      try {
        await video.play();
      } catch(error){
        hideBuffer();

        if(
          error?.name !== 'NotAllowedError' &&
          error?.name !== 'AbortError'
        ){
          showMessage(
            'Unable to play this server. Please select another server.'
          );
        }
      }

      if(video.paused) hideBuffer();

    } catch(error){
      if(requestId !== switchId) return;

      isSwitching = false;
      hideBuffer();

      console.error('Stream load failed:', error);

      showMessage(
        'Unable to load this server. Please select another server.'
      );

    } finally {
      clearTimeout(loadTimer);
    }
  });

  return switchQueue;
}

async function initPlayer(){
  if(!window.shaka){
    showMessage('Video player failed to load.');
    return false;
  }

  shaka.polyfill.installAll();

  if(!shaka.Player.isBrowserSupported()){
    console.error('Browser not supported');

    showMessage(
      'This browser does not support the video player.'
    );

    return false;
  }

  player = new shaka.Player();

  await player.attach(video);

  ui = new shaka.ui.Overlay(player, container, video);

  ui.configure({
    addBigPlayButton: true,

    controlPanelElements: [
      'play_pause',
      'mute',
      'time_and_duration',
      'spacer',
      'language',
      'captions',
      'quality',
      'picture_in_picture',
      'fullscreen'
    ],

    volumeBarColors: {
      base: 'rgba(0,136,255,.3)',
      level: 'rgb(0,136,255)'
    },

    seekBarColors: {
      base: '#828282',
      buffered: '#ffc336',
      played: '#1fc7ff'
    }
  });

  player.addEventListener('buffering', event => {
    if(isSwitching) return;

    if(event.buffering && !video.paused){
      showBuffer();
    } else if(!event.buffering && video.readyState >= 3){
      hideBuffer();
    }
  });

  configureStreaming();

  player.addEventListener('error', event => {
    const error = event.detail;

    console.error('Shaka Player Error:', error);

    if(isSwitching) return;

    if(error?.severity === shaka.util.Error.Severity.CRITICAL){
      hideBuffer();

      showMessage(
        'Unable to play this server. Please select another server.'
      );
    }
  });

  video.addEventListener('error', () => {
    if(!isSwitching){
      hideBuffer();

      showMessage(
        'Unable to play this server. Please select another server.'
      );
    }
  });

  video.addEventListener('playing', hideMessage);

  return true;
}

async function init(){
  if(!slug){
    showMessage('No match selected. Open a match from All Sports.');

    grid.innerHTML = '<div class="channels-empty">Missing match identifier.</div>';

    return;
  }

  try {
    const response = await fetch(API, {
      headers: {
        Accept: 'application/json'
      }
    });

    if(!response.ok){
      throw Error(`HTTP ${response.status}`);
    }

    const data = await response.json();

    const list = Array.isArray(data)
      ? data
      : Array.isArray(data.matches)
        ? data.matches
        : Array.isArray(data.data)
          ? data.data
          : [];

    const match = list.find(m => String(m.slug) === slug);

    if(!match){
      showMessage('Match not found.');

      grid.innerHTML = '<div class="channels-empty">This match is not available.</div>';

      return;
    }

    document.title = `${match.event?.name || match.title || 'Live Match'} | MatchDekho`;

    streams = getStreams(match);

    renderChannels();

    if(!streams.length){
      showMessage(
        'No playable streams available for this match.'
      );
      return;
    }

    const ready = await initPlayer();

    if(ready){
      await switchChannel(0);
    }

  } catch(error){
    console.error('Match loading error', error);

    showMessage(
      'Could not load match details. Please try again.'
    );

    grid.innerHTML = '<div class="channels-empty">Unable to load servers.</div>';
  }
}
